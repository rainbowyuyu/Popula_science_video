function opacity(object:T.Object3D|undefined):number{
 if(!object)return 0;let p:T.Object3D|null=object;while(p){if(!p.visible)return 0;p=p.parent;}
 let alpha=0;object.traverse(o=>{if(o instanceof T.Mesh||o instanceof T.Line){for(const m of Array.isArray(o.material)?o.material:[o.material])alpha=Math.max(alpha,m.opacity)}});return alpha;
}
function onObject(object:T.Object3D|undefined,local=V()):Target|undefined{
 if(!object)return;return {point:object.localToWorld(local),opacity:opacity(object)};
}
function vertex(object:T.Mesh|undefined,index:number):Target|undefined{
 if(!object)return;const p=object.geometry.attributes.position;return onObject(object,V().fromBufferAttribute(p,index));
}
function resolve(w:HypertensionWorld,name:string):Target|undefined{
 if(name==='aortaContext')return {point:w.anchors.vessel.clone().add(V(0,-.20,0)),opacity:opacity(w.atlasGroups.aorta)};
 if(name==='aortaWall')return vertex(w.vessel?.mesh,18*41+2);
 if(name==='intima')return vertex(w.vessel?.mesh,53*41+35*41+20);
 if(name==='arterioleWall')return vertex(w.arteriole?.root.getObjectByName('smooth-muscle-wall') as T.Mesh|undefined,32*37+36);
 if(name==='muscle'){
  const bands:T.Mesh[]=[];w.arteriole?.root.traverse(o=>{if(o instanceof T.Mesh&&o.name==='embedded-circumferential-muscle-bundle')bands.push(o)});
  return vertex(bands[7],49+48);
 }
 if(name==='lumen')return onObject(w.arteriole?.root,V(-.3,0,0));
 if(name==='glomerulus')return onObject(w.renalDiagram?.getObjectByName('bowman-capsule'));
 if(name==='tubule')return onObject(w.renalDiagram,V(.69,.19,.04));
 if(name==='reabsorption'){
  const route=w.signals.find(r=>r.mode==='reabsorb');return route?onObject(route.root,route.curve.getPointAt(.55)):undefined;
 }
 if(name==='hormone'){
  const route=w.signals.find(r=>r.mode==='hormone');return route?onObject(route.root,route.curve.getPointAt(.025)):undefined;
 }
 if(name==='lipid')return onObject(w.lipoprotein?.particle);
 if(name==='receptor')return onObject(w.lipoprotein?.root.getObjectByName('ldl-receptor'));
 if(name==='foam')return onObject(w.foamCells[2]);
 if(name==='plaque')return onObject(w.plaqueCore);
 if(name==='thrombus')return onObject(w.plaqueThrombus);
 if(name==='measureCuff'||name==='measureForearm')return w.anchors[name]?{point:w.anchors[name].clone(),opacity:1}:undefined;
 const point=w.anchors[name];return point?{point:point.clone(),opacity:opacity(w.atlasGroups[name])}:undefined;
}

const textWidth=(text:string)=>[...text].reduce((n,c)=>n+(/[\u0000-\u007f]/.test(c)?24:42),0)+34;
export function getAnnotations(w:HypertensionWorld,s:FrameState):Annotation[]{
 const frame=s.time*30,rows:Annotation[]=[];
 for(const spec of annotationSpecs){
  if(frame<=spec.from||frame>=spec.to)continue;
  const target=resolve(w,spec.target);if(!target)continue;
  const cameraPoint=target.point.clone().applyMatrix4(w.camera.matrixWorldInverse);
  const projected=target.point.clone().project(w.camera);
  if(cameraPoint.z>=-w.camera.near||projected.z<=-1||projected.z>=1)continue;
  const ax=(projected.x+1)*960,ay=(1-projected.y)*540;
  const edge=ease(Math.min(ax-50,1870-ax,ay-190,825-ay)/90);
  const alpha=ease((frame-spec.from)/13)*ease((spec.to-frame)/13)*ease(target.opacity/.16)*edge;
  if(alpha<=.001)continue;
  const width=textWidth(spec.text),height=64;
  const proposedX=spec.side>0?ax+(spec.gap??145):ax-(spec.gap??145)-width;
  let x=Math.max(70,Math.min(1850-width,proposedX));
  let y=Math.max(310,Math.min(690,ay+spec.dy));
  // Reserve the upper right chart area and the permanent lower caption lanes.
  if(x+width>1215&&y<495&&['h013','h019','h020','h039','h040'].some(id=>{const c=timeline.clips.find(c=>c.id===id)!;return frame>=c.start&&frame<c.end}))y=525;
  for(const previous of rows){if(x<previous.x+previous.width+30&&x+width>previous.x-30&&y<previous.y+height+25&&y+height>previous.y-25)y=Math.min(690,previous.y+height+30);}
  rows.push({id:spec.id,text:spec.text,alpha,color:spec.color??'#467c7d',anchor:{x:ax,y:ay},x,y,width,height,side:spec.side});
 }
 return rows.slice(0,2);
}

export function drawAnnotations(canvas:HTMLCanvasElement,w:HypertensionWorld,s:FrameState){
 const ctx=canvas.getContext('2d')!;ctx.clearRect(0,0,1920,1080);
 for(const a of getAnnotations(w,s)){
  const endX=a.side>0?a.x:a.x+a.width,endY=a.y+a.height/2;
  const dx=endX-a.anchor.x,dy=endY-a.anchor.y;
  ctx.save();ctx.globalAlpha=a.alpha;ctx.lineCap='round';ctx.lineJoin='round';
  ctx.strokeStyle='#f2eee7';ctx.lineWidth=6;
  const path=()=>{ctx.beginPath();ctx.moveTo(a.anchor.x,a.anchor.y);ctx.bezierCurveTo(a.anchor.x+dx*.38,a.anchor.y+dy*.04,endX-dx*.18,endY,endX,endY);ctx.stroke()};path();
  ctx.strokeStyle=a.color;ctx.lineWidth=2.1;path();
  ctx.beginPath();ctx.arc(a.anchor.x,a.anchor.y,5,0,Math.PI*2);ctx.fillStyle=a.color;ctx.fill();ctx.lineWidth=2;ctx.strokeStyle='#f2eee7';ctx.stroke();
  // A small translucent paper backing protects type over faint anatomy.
  ctx.fillStyle='rgba(242,238,231,.92)';ctx.beginPath();ctx.roundRect(a.x,a.y,a.width,a.height,12);ctx.fill();
  ctx.fillStyle='#29484b';ctx.font='600 42px "Microsoft YaHei",sans-serif';ctx.textBaseline='middle';ctx.textAlign='center';ctx.fillText(a.text,a.x+a.width/2,a.y+a.height/2+1);
  ctx.restore();
 }
}
